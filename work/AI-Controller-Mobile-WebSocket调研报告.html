4. 技术实现方案 Technical Implementation

Flask-SocketIO Mobile Server - 核心代码
# mobile-server/app.py from flask import Flask, render_template, jsonify from flask_socketio import SocketIO, emit from ai_controller.connectors import HTTPConnector import threading, time app = Flask(__name__) app.config["SECRET_KEY"] = "aictrl-mobile-secret" socketio = SocketIO(app, cors_allowed_origins="*", async_mode="threading") def monitor_loop(): # 监控线程：定时从控制器拉取数据，推送给手机 while True: try: conn = HTTPConnector("192.168.1.100", port=8080) points = conn.get_points(["泵1运行", "水位", "温度"]) socketio.emit("monitor_update", {"data": points, "ts": time.time()}) except Exception as e: socketio.emit("monitor_error", {"msg": str(e)}) time.sleep(1) @socketio.on("connect") def on_connect(): emit("system_info", {"version": "1.0.0", "controller": "192.168.1.100"}) @socketio.on("subscribe_monitor") def on_subscribe_monitor(data): emit("monitor_started", {"interval": data.get("interval", 1)}) @socketio.on("request_config") def on_request_config(data): conn = HTTPConnector("192.168.1.100", port=8080) val = conn.get_config(data["key"]) emit("config_response", {"key": data["key"], "value": val}) @socketio.on("set_config") def on_set_config(data): conn = HTTPConnector("192.168.1.100", port=8080) result = conn.set_config(data["key"], data["value"]) emit("config_updated", {"key": data["key"], "result": result}) socketio.emit("config_changed", {"key": data["key"], "value": data["value"]}) @socketio.on("deploy_config") def on_deploy_config(data): conn = HTTPConnector("192.168.1.100", port=8080) result = conn.deploy(data["config"], data.get("hot_reload", True)) emit("deploy_result", result) threading.Thread(target=monitor_loop, daemon=True).start() if __name__ == "__main__": socketio.run(app, host="0.0.0.0", port=8081, debug=False)

// app.js - 移动端 WebSocket 客户端 import { io } from "https://cdn.socket.io/4.5.4/socket.io.esm.min.js" const socket = io("http://192.168.1.100:8081") socket.on("connect", () => { showToast("已连接") socket.emit("subscribe_monitor", { interval: 1 }) }) socket.on("monitor_update", (payload) => { payload.data.forEach(pt => { const card = document.getElementById("point-" + pt.alias) if (card) card.querySelector(".point-val").textContent = pt.value + pt.unit chart.data.labels.push(new Date().toLocaleTimeString()) chart.data.datasets.find(d => d.label === pt.alias).data.push(pt.value) if (chart.data.labels.length > 30) { chart.data.labels.shift() chart.data.datasets.forEach(d => d.data.shift()) } }) chart.update() }) socket.on("log_entry", (payload) => { const el = document.getElementById("log-viewer") el.innerHTML += payload.formatted + "
" el.scrollTop = el.scrollHeight })
